const {test} = require('@playwright/test')

test('Seach for Playwright Automation in Google', async({page}) =>{
    await page.goto("https://www.google.com", {waitUntil:'domcontentloaded'})

    const AcceptBtn = page.locator('button:has-text("Acccept all")');
    if(await AcceptBtn.isVisible()){
        await AcceptBtn.click();
    }

    //Find the search box named q
    const searchBox = page.locator('textarea[name="q"]')

    //Type Playwright Automation in the search box
    await searchBox.fill("Playwright Automation")

    //Press the Enter Button
    await searchBox.press('Enter')

    //wait for the result page to be loaded
    await page.waitForLoadState('networkidle')

    //Wait for 5 seconds to close the browser
    await page.waitForTimeout(5000)
}
)
